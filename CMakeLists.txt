cmake_minimum_required(VERSION 3.24)
project(Neon VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Test)
qt_standard_project_setup(REQUIRES 6.8)
find_package(ZLIB REQUIRED)
add_library(neon_core src/store.cpp src/formats.cpp)
target_include_directories(neon_core PUBLIC src)
target_link_libraries(neon_core PUBLIC Qt6::Core Qt6::Gui ZLIB::ZLIB)
qt_add_executable(neon MACOSX_BUNDLE src/main.cpp src/platform.cpp)
target_link_libraries(neon PRIVATE neon_core Qt6::Quick Qt6::QuickControls2 Qt6::Network)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(neon PRIVATE src/platform_mac.mm)
  target_link_libraries(neon PRIVATE "-framework AppKit" "-framework Security")
endif()
qt_add_qml_module(neon URI Neon VERSION 1.0 QML_FILES qml/Main.qml SOURCES src/controller.h src/controller.cpp)
set_target_properties(neon PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.johnhenning.neon MACOSX_BUNDLE_BUNDLE_NAME Neon MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} OUTPUT_NAME Neon)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(neon_tests tests/test_core.cpp)
  target_link_libraries(neon_tests PRIVATE neon_core Qt6::Test)
  add_test(NAME core COMMAND neon_tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME ui_smoke COMMAND neon --smoke-test)
endif()
install(TARGETS neon BUNDLE DESTINATION . RUNTIME DESTINATION bin)
