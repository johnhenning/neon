cmake_minimum_required(VERSION 3.24)
project(NeonPortableCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(neon_portable_core src/workspace.cpp)
target_include_directories(neon_portable_core PUBLIC include)
target_compile_options(neon_portable_core PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(workspace_test tests/workspace_test.cpp)
target_link_libraries(workspace_test PRIVATE neon_portable_core)
add_test(NAME workspace COMMAND workspace_test)
